rubik.hs

{-# LANGUAGE CPP, TemplateHaskell #-}
import Rapids

cutBoth :: (Solid -> Solid) -> (Solid -> Solid) -> E V3 -> Solid
cutBoth a b ax = a (translate ax (-1) centeredCube) + b (translate ax 1 centeredCube)
gaps = $black $ translated ez (1/3) (translate ez (1/3 - 0.5) (scale 1 0.02 centeredCube))


solved = 
    centeredCube
      - cutBoth $white $yellow ez
      - cutBoth $red $green ex
      - cutBoth $blue $purple ey
      - rotatedDeg ex 90 gaps
      - rotateDeg ey 90 gaps

scrambled = l $ u $ r $ u' $ r $ u $ l solved

split :: E V3 -> Lens' Solid (Solid,Solid,Solid)
split (E ax) f a = f (a * translate (E ax) (1/3) centeredCube,
                      a * scale (E ax) (1/3) centeredCube,
                      a * translate (E ax) (-1/3) centeredCube)
  <&> \(a,b,c) -> a+b+c

d  s = s & split ez . _1 %~ rotate ez (pi/2)
u  s = s & split ez . _3 %~ rotate ez (pi/2)
r  s = s & split ex . _3 %~ rotate ex (pi/2)
l  s = s & split ex . _1 %~ rotate ex (pi/2)
f  s = s & split ey . _3 %~ rotate ey (pi/2)
b  s = s & split ey . _1 %~ rotate ey (pi/2)
d' s = s & split ez . _1 %~ rotate ez (-pi/2)
u' s = s & split ez . _3 %~ rotate ez (-pi/2)
r' s = s & split ex . _3 %~ rotate ex (-pi/2)
l' s = s & split ex . _1 %~ rotate ex (-pi/2)
f' s = s & split ey . _3 %~ rotate ey (-pi/2)
b' s = s & split ey . _1 %~ rotate ey (-pi/2)

main =
  writeSTEPColor (takeWhile (/= '.') __FILE__ ++ ".step") $ scrambled + translate ex 2 solved

sweepRuled.hs

{-# LANGUAGE CPP #-}
import Rapids
import Waterfall.SVG

main = do
  Right s <- readSVG "king.svg"
  writeSTEPColor (takeWhile (/= '.') __FILE__ ++ ".step") $ spikes s

spikes s = unitPolygon 3
  & sweepRuled f
  & rotated
      ez (pi/3)
      ez (pi/6)
  & rotate ey (pi/2)
  where f v = rotate
                ez (unangle v)
                ex (pi/2)
                (foldMap toPath s)

unitSpiral.hs

{-# LANGUAGE CPP, TemplateHaskell #-}
import Data.List
import Rapids

main = do
  writeSTEPColor (takeWhile (/= '.') __FILE__ ++ ".step") $
    foldr1 (stacked ex) $
      intersperse
        unitGap
        [ $green $ unitSpiral 2 $ rectangle 0.5 0.2,
          $red $ offset 0.05 [5, 6] $ mirrorZ1 taperedSpiral,
          fan
        ]

fan = $blue
  (unions
    [ rotate ez th ex (pi / 4) $ revolution bladeAngle $ rectangle 2 0.2
    | j <- [0 .. n - 1],
      let th = 2 * pi * fromIntegral j / fromIntegral n
    ])
    + $yellow (scale 0.5 0.3 unitCylinder)
    - $brown (scale 0.3 1 unitCylinder)
  where
    n = 4
    cover = 0.5
    bladeAngle = cover * 2 * pi / fromIntegral n

mirrorZ1 = _translated ez (-1) %~ mirror ez

taperedSpiral = unitSpiral 2 0.2 $ rectangle 0.5 0.2
